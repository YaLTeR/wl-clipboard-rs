5fn from_u8_singlerule(bytes: &[u8], rule: &super::MagicRule) -> bool {
6    // Check if we're even in bounds
7    let bound_min = rule.start_off as usize;
8    if bound_min > bytes.len() {
9        return false;
10    }
11    let bound_max = rule.start_off as usize + rule.val.len() + rule.region_len as usize;
12    let bound_max = min(bound_max, bytes.len());
13
14    let testarea = &bytes[bound_min..bound_max];
15
16    testarea.windows(rule.val.len()).any(|window| {
17        // Apply mask to value
18        match rule.mask {
19            None => rule.val == window,
20            Some(mask) => {
21                assert_eq!(window.len(), mask.len());
22                let masked = zip(window, mask).map(|(a, b)| a & b);
23                rule.val.iter().copied().eq(masked)
24            }
25        }
26    })
27}

53    }
54
55    // Check subrules recursively
56    for y in n {
57        let rule = &graph[y];
58
59        if from_u8_singlerule(bytes, rule) {
60            // Check next indent level if needed
61            if graph.neighbors_directed(y, Outgoing).count() != 0 {
62                return from_u8_walker(bytes, graph, y, false);
63            // Next indent level is lower, so this must be it
64            } else {
65                return true;
66            }
67        }
68    }
